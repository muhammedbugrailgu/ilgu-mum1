            if (type === 'jar') {

                document.getElementById('custom-svg-wax-top').setAttribute('fill', visualValue);

            const total = customCandle.basePrice + customCandle.jarExtra + customCandle.waxExtra + customCandle.wickExtra;

        // --- 6. ALIŞVERİŞ SEPETİ YÖNETİMİ ---
        let cart = [
            {
                id: 'ilgu-01',
                name: 'Sandal Ağacı & Kehribar',
                details: '220g Soya Mumu • 50 Saat',
                price: 380,
                qty: 1
            }
        ];

        function addToCart(productId) {
            const product = products.find(p => p.id === productId);
            if (!product) return;

            const existing = cart.find(item => item.id === productId);
            if (existing) {
                existing.qty += 1;
            } else {
                cart.push({
                    id: product.id,
                    name: product.name,
                    details: `${product.weight} Soya Mumu • ${product.burnTime}`,
                    price: product.price,
                    qty: 1
                });
            }
            updateCartUI();
            showToast(`"${product.name}" sepete eklendi`);
        }

        function changeQty(itemId, delta) {
            const idx = cart.findIndex(item => item.id === itemId);
            if (idx === -1) return;

            cart[idx].qty += delta;
            if (cart[idx].qty <= 0) {
                cart.splice(idx, 1);
            }
            updateCartUI();
        }

        function updateCartUI() {
            const container = document.getElementById('cart-items');
            const badge = document.getElementById('cart-badge');
            const subtotalEl = document.getElementById('cart-subtotal');
            const shippingEl = document.getElementById('cart-shipping');
            const totalEl = document.getElementById('cart-total');
            const shipText = document.getElementById('free-shipping-text');
            const shipBar = document.getElementById('free-shipping-bar');

            const totalCount = cart.reduce((acc, item) => acc + item.qty, 0);
            badge.textContent = totalCount;

            if (cart.length === 0) {
                container.innerHTML = `
                    <div class="h-64 flex flex-col items-center justify-center text-center opacity-65">
                        <i class="ph ph-flame text-5xl mb-3"></i>
                        <p class="font-serif text-xl">Sepetiniz şu an boş</p>
                        <p class="text-xs mt-1">İLGÜ koleksiyonumuzdan ruhunuza hitap eden bir koku seçin.</p>
                    </div>
                `;
            } else {
                container.innerHTML = cart.map(item => `
                    <div class="p-4 rounded-2xl bg-wood-100/80 dark:bg-wood-800/70 border border-wood-700/15 flex items-center justify-between gap-4">
                        <div class="space-y-1">
                            <h4 class="font-serif font-bold text-lg leading-tight">${item.name}</h4>
                            <p class="text-xs opacity-75">${item.details}</p>
                            <p class="text-sm font-bold text-amber-800 dark:text-amber-400">${item.price * item.qty} ₺</p>
                        </div>
                        <div class="flex items-center gap-2 bg-wood-50 dark:bg-wood-900 px-2.5 py-1.5 rounded-full border border-wood-700/20">
                            <button onclick="changeQty('${item.id}', -1)" class="w-6 h-6 rounded-full hover:bg-wood-200 dark:hover:bg-wood-700 flex items-center justify-center text-xs font-bold">-</button>
                            <span class="text-xs font-semibold w-4 text-center">${item.qty}</span>
                            <button onclick="changeQty('${item.id}', 1)" class="w-6 h-6 rounded-full hover:bg-wood-200 dark:hover:bg-wood-700 flex items-center justify-center text-xs font-bold">+</button>
                        </div>
                    </div>
                `).join('');
            }

            const subtotal = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
            const freeLimit = 750;
            const shipping = (subtotal === 0 || subtotal >= freeLimit) ? 0 : 65;

            subtotalEl.textContent = `${subtotal} ₺`;
            shippingEl.textContent = shipping === 0 ? 'Ücretsiz' : `${shipping} ₺`;
            totalEl.textContent = `${subtotal + shipping} ₺`;

            const progress = Math.min(100, (subtotal / freeLimit) * 100);
            shipBar.style.width = `${progress}%`;
            if (subtotal >= freeLimit) {
                shipText.innerHTML = '🎉 <strong>Tebrikler!</strong> Ücretsiz kargo ve kibrit şişesi hediyeniz aktif.';
            } else {
                shipText.innerHTML = `Ücretsiz kargo için <strong>${freeLimit - subtotal} ₺</strong> değerinde daha ürün ekleyin.`;
            }
        }

        function toggleCart(open) {
            const overlay = document.getElementById('cart-overlay');
            const drawer = document.getElementById('cart-drawer');
            if (open) {
                overlay.classList.remove('hidden');
                setTimeout(() => overlay.classList.remove('opacity-0'), 10);
                drawer.classList.remove('translate-x-full');
            } else {
                overlay.classList.add('opacity-0');
                drawer.classList.add('translate-x-full');
                setTimeout(() => overlay.classList.add('hidden'), 300);
            }
        }

        function checkoutOrder() {
            if (cart.length === 0) {
                showToast('Sipariş vermek için sepetinize ürün ekleyin.');
                return;
            }
            cart = [];
            updateCartUI();
            toggleCart(false);
            showToast('Siparişiniz alındı! İLGÜ Atölyesi mumlarınızı özenle hazırlıyor 🕯️✨');
        }

        // --- 7. TOAST BİLDİRİM FONKSİYONU ---
        let toastTimeout;
        function showToast(message) {
            const toast = document.getElementById('toast');
            const text = document.getElementById('toast-message');
            text.textContent = message;
            toast.classList.remove('opacity-0', 'translate-y-4');
            clearTimeout(toastTimeout);
            toastTimeout = setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-4');
            }, 3200);
        }
